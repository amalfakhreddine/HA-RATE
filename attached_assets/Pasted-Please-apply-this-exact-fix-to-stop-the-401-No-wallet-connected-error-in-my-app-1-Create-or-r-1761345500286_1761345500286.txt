1️⃣ Create or replace this React file:
📁 src/components/ClaimButton.jsx
────────────────────────────
import React from "react";
import { useTonConnect } from "@tonconnect/ui-react";

export default function ClaimButton() {
  const { connector, account } = useTonConnect() || {};

  const handleClaim = async () => {
    try {
      if (!account?.address) {
        alert("Please connect your wallet first.");
        return;
      }

      let signed = null;
      try {
        const message = `Claim request:${Date.now()}`;
        signed = await connector?.request({
          method: "ton_sign",
          params: { address: account.address, data: message },
        });
      } catch {
        signed = null;
      }

      const headers = {
        "Content-Type": "application/json",
        "x-wallet-address": account.address,
      };
      if (signed) headers["x-wallet-signature"] = JSON.stringify(signed);

      const res = await fetch("/api/claim", {
        method: "POST",
        headers,
        body: JSON.stringify({ claimType: "mining" }),
      });

      const json = await res.json();
      if (!res.ok) return alert(`Claim failed: ${json?.error || res.status}`);

      alert(`Claim success: ${JSON.stringify(json)}`);
    } catch (err) {
      console.error(err);
      alert("Claim error, check console.");
    }
  };

  return (
    <button onClick={handleClaim} className="btn-claim px-4 py-2 rounded">
      Claim Rewards
    </button>
  );
}
